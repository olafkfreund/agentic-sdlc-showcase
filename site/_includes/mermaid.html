  async function render() {
    const theme = document.documentElement.getAttribute('data-theme') === 'light' ? 'light' : 'dark';
    mermaid.initialize({
      startOnLoad: false,
      theme: 'base',
      securityLevel: 'strict',
      fontFamily: 'ui-monospace, "JetBrains Mono", "Fira Code", Menlo, Consolas, monospace',
      themeVariables: { fontSize: '14px', ...PALETTES[theme] },
    });
    // Restore the source and clear mermaid's done-marker, or run() skips the element and the
    // diagram keeps the colours of the theme it was first drawn in.
    document.querySelectorAll('pre.mermaid').forEach((el) => {
      el.textContent = el.dataset.src;
      el.removeAttribute('data-processed');
    });
    // A diagram that fails to parse must not take the page down with it, and must say so
    // rather than leaving a silent gap — the argument the gates make about passing quietly.
    try {
      await mermaid.run({ querySelector: 'pre.mermaid' });
    } catch (err) {
      console.error('[mermaid] diagram failed to render:', err);
    }
  }

  await render();
  document.addEventListener('themechange', render);
